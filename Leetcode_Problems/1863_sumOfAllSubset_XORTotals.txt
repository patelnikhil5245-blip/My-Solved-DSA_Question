class Solution {
    
    public int recur(int[] nums, int i, int xor) {

        if (i == nums.length) {
            return xor;
        }

        return recur(nums, i + 1, xor ^ nums[i]) +  recur(nums, i + 1, xor);
    }

    public int subsetXORSum(int[] nums) {

    }
}
